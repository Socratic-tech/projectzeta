How to add Sidebar.html in Google Apps Script

  1. 1
    Open the school's existing Google Sheet.
    In Google Sheets, choose Extensions → Apps Script. This is the Extensions menu inside Google Sheets, not the browser Extensions menu.
  2. 2
    Create the Sidebar HTML file if it does not already exist.
    In the Apps Script editor, look at the Files list on the left. Click the + button beside Files, then choose HTML.
  3. 3
    Name the file exactly Sidebar.
    Do not type Sidebar.html. Apps Script automatically adds the .html extension, so the file will appear as Sidebar.html.
  4. 4
    Delete everything Apps Script put in the new file.
    The new HTML file usually contains starter HTML. Select all of it and delete it.
  5. 5
    Copy the complete code below and paste it into Sidebar.html.
    Use the Copy Sidebar Code button. Then paste into the empty Sidebar file in Apps Script.
  6. 6
    Save the Apps Script project.
    Then return to the Google Sheet and refresh/reopen it. The Vocational Tracker menu should appear in Sheets.
Existing Sidebar file? Click Sidebar.html in the Files list, select everything, delete it, and replace it with the code below. You do not need to create a second Sidebar file.

    $("busy").textContent = message || "Working…";

    setBusy(false);

  }

    $("revealValue").textContent = value;
    $("revealNote").textContent =
      note || "Copy this now. This credential will not be shown again.";
    $("reveal").classList.add("show");

    window.scrollTo({ top: 0, behavior: "smooth" });
  }

  function copyReveal() {
    if (!currentReveal) return;

    navigator.clipboard.writeText(currentReveal).then(function () {
      var original = $("revealNote").textContent;
      $("revealNote").textContent = "Copied to clipboard.";
      setTimeout(function () {
        $("revealNote").textContent = original;
      }, 1600);
    }).catch(function () {
      var value = $("revealValue");
      var range = document.createRange();
      range.selectNodeContents(value);
      var sel = window.getSelection();
      sel.removeAllRanges();
      sel.addRange(range);
    });
  }

  function initials(name) {
    return String(name || "?")
      .split(/\s+/)
      .filter(Boolean)
      .slice(0, 2)
      .map(function (part) { return part.charAt(0).toUpperCase(); })
      .join("") || "?";
  }

  function escapeHtml(value) {
    return String(value == null ? "" : value)
      .replace(/&/g, "&amp;")
      .replace(/</g, "&lt;")
      .replace(/>/g, "&gt;")
      .replace(/"/g, "&quot;")
      .replace(/'/g, "&#039;");
  }

  function jsString(value) {
    return String(value == null ? "" : value)
      .replace(/\\/g, "\\\\")
      .replace(/'/g, "\\'");
  }

  function refresh() {
    setBusy(true, "Refreshing…");

    google.script.run
      .withSuccessHandler(function (st) {
        setBusy(false);
        state = st || {};
        render(state);
      })
      .withFailureHandler(fail)
      .uiStatus();
  }

  function render(st) {
    var admins = Array.isArray(st.admins) ? st.admins : [];
    var viewers = Array.isArray(st.viewers) ? st.viewers : [];

    var activeAdmins = admins.filter(function (a) { return a.active; }).length;
    var activeViewers = viewers.filter(function (v) { return v.active; }).length;

    $("statAdmins").innerHTML =
      activeAdmins
        ? '<span class="good">' + activeAdmins + ' active</span>'
        : '<span class="warn">none</span>';

    $("statViewers").innerHTML =
      activeViewers
        ? '<span class="good">' + activeViewers + ' active</span>'
        : '<span class="warn">none</span>';

    $("statToken").innerHTML =
      st.entryToken
        ? '<span class="good">ready</span>'
        : '<span class="danger-text">missing</span>';

    $("statSetup").innerHTML =
      st.entryToken && activeAdmins > 0
        ? '<span class="good">ready</span>'
        : '<span class="warn">finish setup</span>';

    $("statVersion").innerHTML =
      '<span class="good">v' + escapeHtml(st.release || "3.0.0") +
      ' · schema ' + escapeHtml(st.schemaVersion || 3) + '</span>';

    $("migrationNotice").style.display =
      st.legacyAdminPresent && activeAdmins === 0 ? "block" : "none";

    $("tokenPreview").textContent =
      st.entryToken || "No entry token is set.";

    $("publicBaseUrl").value = st.publicSiteBaseUrl || "";
    $("supportEmail").value = st.supportEmail || "";

    if (st.entryPageUrl) {
      $("entryUrlBox").style.display = "block";
      $("entryUrlPreview").textContent = st.entryPageUrl;
    } else {
      $("entryUrlBox").style.display = "none";
      $("entryUrlPreview").textContent = "";
    }

    $("firstAdminCard").style.display =
      activeAdmins === 0 ? "block" : "none";

    renderAdmins(admins);
    renderViewers(viewers);
  }

  function renderAdmins(admins) {
    if (!admins.length) {
      $("admins").innerHTML =
        '<div class="empty">No personal administrators have been created yet.</div>';
      return;
    }

    $("admins").innerHTML = admins.map(function (a) {
      var safeName = escapeHtml(a.name);
      var safeId = jsString(a.id);

      return (
        '<div class="person' + (a.active ? '' : ' off') + '">' +
          '<div class="avatar admin">' + escapeHtml(initials(a.name)) + '</div>' +
          '<div class="person-main">' +
            '<div class="person-name">' + safeName + '</div>' +
            '<div class="person-sub">' +
              (a.active
                ? '<span class="pill admin">Admin + Reports</span>'
                : '<span class="pill warn">Disabled</span>') +
            '</div>' +
            '<div class="person-sub">' +
              (a.email ? escapeHtml(a.email) : '<strong>No email set</strong>') +
            '</div>' +
          '</div>' +
          '<div class="person-actions">' +
            '<button class="btn-secondary btn-small" onclick="regenerateAdmin(\'' + safeId + '\', \'' + jsString(a.name) + '\')">New code</button>' +
            '<button class="btn-secondary btn-small" onclick="editAdminEmail(\'' + safeId + '\', \'' + jsString(a.email || "") + '\')">' +
              (a.email ? 'Edit email' : 'Set email') +
            '</button>' +
            '<button class="' + (a.active ? 'btn-danger' : 'btn-secondary') + ' btn-small" ' +
              'onclick="setAdminActive(\'' + safeId + '\', ' + (!a.active) + ')">' +
              (a.active ? 'Disable' : 'Enable') +
            '</button>' +
          '</div>' +
        '</div>'
      );
    }).join("");
  }

  function renderViewers(viewers) {
    if (!viewers.length) {
      $("viewers").innerHTML =
        '<div class="empty">No report-only viewers have been added.</div>';
      return;
    }

    $("viewers").innerHTML = viewers.map(function (v) {
      var safeName = escapeHtml(v.name);
      var safeCode = escapeHtml(v.code || "");
      var safeId = jsString(v.id || v.code || "");

      return (
        '<div class="person' + (v.active ? '' : ' off') + '">' +
          '<div class="avatar">' + escapeHtml(initials(v.name)) + '</div>' +
          '<div class="person-main">' +
            '<div class="person-name">' + safeName + '</div>' +
            '<div class="person-sub">' +
              (v.active
                ? '<span class="pill good">Reports</span> <span class="mono">' + safeCode + '</span>'
                : '<span class="pill warn">Disabled</span> <span class="mono">' + safeCode + '</span>') +
            '</div>' +
            '<div class="person-sub">' +
              (v.email ? escapeHtml(v.email) : '<strong>No email set</strong>') +
            '</div>' +
          '</div>' +
          '<div class="person-actions">' +
            '<button class="btn-secondary btn-small" onclick="emailViewerCode(\'' + safeId + '\')">Email code</button>' +
            '<button class="btn-secondary btn-small" onclick="editViewerEmail(\'' + safeId + '\', \'' + jsString(v.email || "") + '\')">' +
              (v.email ? 'Edit email' : 'Set email') +
            '</button>' +
            '<button class="' + (v.active ? 'btn-danger' : 'btn-secondary') + ' btn-small" ' +
              'onclick="setViewerActive(\'' + safeId + '\', ' + (!v.active) + ')">' +
              (v.active ? 'Disable' : 'Enable') +
            '</button>' +
          '</div>' +
        '</div>'
      );
    }).join("");
  }

  function credentialNote(result, accessText) {
    if (result && result.emailSent) {
      return "The code was emailed to " + result.email +
        ". The tracker cannot recover the plaintext code later.";
    }

    var problem = result && result.emailError
      ? " Email delivery failed: " + result.emailError
      : "";

    return "Save this code now. It grants " + accessText +
      " and cannot be recovered from the tracker." + problem;
  }

  function createFirstAdmin() {
    var name = $("firstAdminName").value.trim();
    var email = $("firstAdminEmail").value.trim();

    google.script.run
      .withSuccessHandler(function (result) {
        setBusy(false);

        $("firstAdminEmail").value = "";

          result.code,
          credentialNote(result, "Admin and Reports access")

        refresh();
      })

      .uiCreateFirstAdmin(name, email);
  }
